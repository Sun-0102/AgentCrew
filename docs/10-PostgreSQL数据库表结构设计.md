# PostgreSQL 数据库表结构设计（多端数据共享版）

本文档基于 `src-tauri/migrations/001_*.sql` ~ `016_*.sql` 全部现有 SQLite 表结构，结合 [09-PostgreSQL多端数据共享迁移方案.md](./09-PostgreSQL多端数据共享迁移方案.md) 中的多端改造要求，给出完整、可直接执行的 PostgreSQL 建表 DDL 与设计说明。

---

## 1. 从 SQLite 到 PostgreSQL 的关键映射说明

| 维度 | 原 SQLite 写法 | PostgreSQL 目标写法 | 说明 |
| :--- | :--- | :--- | :--- |
| 自增主键 | `INTEGER PRIMARY KEY AUTOINCREMENT` | `BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY` | 现代 PG 标准语法，兼容显式导入历史序列号 |
| 时间戳 | `TEXT DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))` | `TIMESTAMPTZ NOT NULL DEFAULT NOW()` | 原 `codex_message_timestamp` 触发器不再需要 |
| 布尔值 | `INTEGER` (`0` / `1`) | `BOOLEAN` (`FALSE` / `TRUE`) | 涉及 `group_sessions.archived`、`agents.enabled`、`app_controls.paused` |
| JSON 字段 | `TEXT` | `JSONB` | 涉及 `chat_groups.members`、`agents.capabilities`、`group_task_steps` 快照与消息 |
| 运行态隔离 | 单机无设备标识 | 新增 `devices` 表及各运行态表的 `device_id` | 避免 B 端启动 `recover` 误杀 A 端正在运行的任务 |
| 项目路径 | `projects.primary_root TEXT NOT NULL UNIQUE` | 拆分为 `project_device_roots(project_id, device_id, local_root)` | 解决不同电脑本地代码仓库路径不一致问题 |
| 单例控制表 | `app_controls(id=1)` | `app_controls(device_id PK)` | 每个客户端设备独立维护暂停与实例状态 |
| 活跃任务唯一索引 | `ON group_tasks((1))` 全局唯一 | `ON group_tasks(device_id)` 按设备唯一 | 允许不同设备各自并发执行群组任务 |
| 跨端附件 | 仅存本机磁盘路径 | 新增 `attachments` 表（`BYTEA` + `sha256`） | 客户端按需拉取并缓存到本机 `app_data_dir` |

---

## 2. 完整建表 SQL（PostgreSQL 16+ / 18.6）

> 可直接在目标库（如 `agentcrew`）中执行以下完整脚本。

```sql
-- ============================================================================
-- AgentCrew PostgreSQL Schema (Multi-Device Shared Storage)
-- 基于 SQLite migrations 001 ~ 016 + 多端共享扩展
-- ============================================================================

-- ----------------------------------------------------------------------------
-- 1. 设备与项目路径映射（多端核心基础表）
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS devices (
    id           TEXT PRIMARY KEY,
    name         TEXT NOT NULL,
    platform     TEXT NOT NULL, -- 'macos' | 'windows' | 'linux'
    last_seen_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS projects (
    id              TEXT PRIMARY KEY,
    name            TEXT NOT NULL,
    repo_remote_url TEXT,
    -- 兼容字段：多端模式下真实本机路径以 project_device_roots 为准
    primary_root    TEXT,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS project_device_roots (
    project_id TEXT NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    device_id  TEXT NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
    local_root TEXT NOT NULL,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (project_id, device_id)
);

-- ----------------------------------------------------------------------------
-- 2. 会话与群组基础表
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS conversations (
    id             TEXT PRIMARY KEY,
    project_id     TEXT UNIQUE REFERENCES projects(id) ON DELETE SET NULL,
    workspace_path TEXT NOT NULL DEFAULT ''
);

CREATE TABLE IF NOT EXISTS chat_groups (
    conversation_id TEXT PRIMARY KEY REFERENCES conversations(id) ON DELETE CASCADE,
    name            TEXT NOT NULL,
    members         JSONB NOT NULL DEFAULT '[]'::jsonb,
    project_id      TEXT REFERENCES projects(id) ON DELETE SET NULL,
    parent_group_id TEXT REFERENCES chat_groups(conversation_id) ON DELETE SET NULL,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS group_sessions (
    conversation_id TEXT PRIMARY KEY REFERENCES chat_groups(conversation_id) ON DELETE CASCADE,
    group_id        TEXT NOT NULL REFERENCES chat_groups(conversation_id) ON DELETE CASCADE,
    name            TEXT NOT NULL,
    archived        BOOLEAN NOT NULL DEFAULT FALSE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_group_sessions_group
    ON group_sessions (group_id, created_at);

-- ----------------------------------------------------------------------------
-- 3. Agent 注册与配对表
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS agents (
    id                 TEXT PRIMARY KEY,
    display_name       TEXT NOT NULL,
    provider_kind      TEXT NOT NULL,
    runtime_kind       TEXT NOT NULL,
    registration_state TEXT NOT NULL DEFAULT 'pending',
    availability       TEXT NOT NULL DEFAULT 'checking',
    enabled            BOOLEAN NOT NULL DEFAULT TRUE,
    source             TEXT NOT NULL DEFAULT 'pairing',
    runtime_version    TEXT,
    executable_path    TEXT,
    capabilities       JSONB NOT NULL DEFAULT '{}'::jsonb,
    error              TEXT,
    created_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at         TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

INSERT INTO agents (id, display_name, provider_kind, runtime_kind, registration_state, source)
VALUES
    ('codex', 'Codex', 'codex', 'codex-app-server', 'active', 'legacy'),
    ('deepseek', 'DeepSeek', 'deepseek', 'deepseek-harness-sdk', 'active', 'legacy')
ON CONFLICT (id) DO NOTHING;

CREATE TABLE IF NOT EXISTS pairing_invitations (
    id               TEXT PRIMARY KEY,
    instance_id      TEXT NOT NULL,
    device_id        TEXT REFERENCES devices(id) ON DELETE SET NULL,
    expected_provider TEXT NOT NULL,
    display_name     TEXT NOT NULL,
    code_hash        TEXT NOT NULL,
    expires_at       BIGINT NOT NULL,
    state            TEXT NOT NULL DEFAULT 'pending',
    failures         INTEGER NOT NULL DEFAULT 0,
    protocol_version INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS pairing_claims (
    id                       TEXT PRIMARY KEY,
    invitation_id            TEXT NOT NULL REFERENCES pairing_invitations(id) ON DELETE CASCADE,
    request_id               TEXT NOT NULL,
    request_fingerprint      TEXT NOT NULL,
    agent_id                 TEXT NOT NULL REFERENCES agents(id) ON DELETE CASCADE,
    token_hash               TEXT NOT NULL,
    token_expiry             BIGINT NOT NULL,
    state                    TEXT NOT NULL DEFAULT 'validating',
    reason_code              TEXT,
    reason                   TEXT,
    reported_provider_kind   TEXT,
    reported_runtime_kind    TEXT,
    verified_provider_kind   TEXT,
    verified_runtime_kind    TEXT,
    verified_runtime_version TEXT,
    UNIQUE (invitation_id, request_id)
);

CREATE TABLE IF NOT EXISTS agent_sessions (
    agent_id                TEXT NOT NULL REFERENCES agents(id) ON DELETE CASCADE,
    conversation_id         TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    runtime_conversation_id TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    workspace_path          TEXT NOT NULL DEFAULT '',
    PRIMARY KEY (agent_id, conversation_id),
    UNIQUE (runtime_conversation_id)
);

-- ----------------------------------------------------------------------------
-- 4. 各 Provider 运行态会话与消息表
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS codex_sessions (
    conversation_id TEXT PRIMARY KEY REFERENCES conversations(id) ON DELETE CASCADE,
    thread_id       TEXT NOT NULL UNIQUE,
    model           TEXT NOT NULL,
    runtime_version TEXT NOT NULL,
    provider        TEXT NOT NULL,
    status          TEXT NOT NULL DEFAULT 'idle',
    turn_id         TEXT,
    error           TEXT,
    device_id       TEXT REFERENCES devices(id) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS codex_messages (
    sequence        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    conversation_id TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    item_id         TEXT NOT NULL,
    role            TEXT NOT NULL,
    text            TEXT NOT NULL,
    agent           TEXT NOT NULL DEFAULT 'codex',
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (conversation_id, item_id)
);

CREATE INDEX IF NOT EXISTS idx_codex_messages_conv_seq
    ON codex_messages (conversation_id, sequence);

CREATE TABLE IF NOT EXISTS deepseek_sessions (
    conversation_id TEXT PRIMARY KEY,
    thread_id       TEXT NOT NULL UNIQUE,
    model           TEXT NOT NULL,
    runtime_version TEXT NOT NULL,
    workspace_path  TEXT NOT NULL DEFAULT '',
    status          TEXT NOT NULL DEFAULT 'idle',
    error           TEXT,
    device_id       TEXT REFERENCES devices(id) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS deepseek_messages (
    sequence   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    item_id    TEXT NOT NULL UNIQUE,
    role       TEXT NOT NULL,
    text       TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS gemini_sessions (
    conversation_id TEXT PRIMARY KEY REFERENCES conversations(id) ON DELETE CASCADE,
    thread_id       TEXT NOT NULL UNIQUE,
    turn_id         TEXT,
    model           TEXT NOT NULL,
    runtime_version TEXT NOT NULL,
    workspace_path  TEXT NOT NULL DEFAULT '',
    status          TEXT NOT NULL DEFAULT 'idle',
    error           TEXT,
    device_id       TEXT REFERENCES devices(id) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS group_message_summaries (
    sequence        BIGINT PRIMARY KEY REFERENCES codex_messages(sequence) ON DELETE CASCADE,
    conversation_id TEXT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    excerpt         TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_group_message_summaries_conv
    ON group_message_summaries (conversation_id, sequence);

-- ----------------------------------------------------------------------------
-- 5. 群组协作任务与调度表
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS group_tasks (
    id                     TEXT PRIMARY KEY,
    conversation_id        TEXT NOT NULL REFERENCES chat_groups(conversation_id) ON DELETE CASCADE,
    device_id              TEXT REFERENCES devices(id) ON DELETE SET NULL,
    objective              TEXT NOT NULL,
    status                 TEXT NOT NULL,
    current_agent          TEXT NOT NULL,
    codex_model            TEXT NOT NULL,
    deepseek_model         TEXT NOT NULL,
    turn_count             INTEGER NOT NULL DEFAULT 0,
    max_turns              INTEGER NOT NULL DEFAULT 7,
    sequence_start         BIGINT NOT NULL DEFAULT 0,
    thread_id              TEXT,
    turn_id                TEXT,
    pending_prompt         TEXT NOT NULL,
    workspace_path         TEXT,
    pending_source_agent   TEXT NOT NULL DEFAULT 'user',
    pending_source_message TEXT,
    generation             INTEGER NOT NULL DEFAULT 0,
    error                  TEXT,
    created_at             TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at             TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_group_tasks_conversation
    ON group_tasks (conversation_id, created_at);

-- 每台设备同时只允许一个活跃群组任务，多设备间互不干扰
CREATE UNIQUE INDEX IF NOT EXISTS idx_one_active_group_task_per_device
    ON group_tasks (device_id)
    WHERE status IN ('running', 'paused', 'cancelling');

CREATE TABLE IF NOT EXISTS group_task_members (
    task_id        TEXT NOT NULL REFERENCES group_tasks(id) ON DELETE CASCADE,
    agent          TEXT NOT NULL REFERENCES agents(id) ON DELETE CASCADE,
    status         TEXT NOT NULL DEFAULT 'idle',
    sequence_start BIGINT NOT NULL DEFAULT 0,
    thread_id      TEXT,
    turn_id        TEXT,
    pending_prompt TEXT NOT NULL DEFAULT '',
    source_agent   TEXT NOT NULL DEFAULT 'user',
    source_message TEXT,
    error          TEXT,
    model          TEXT NOT NULL DEFAULT '',
    PRIMARY KEY (task_id, agent)
);

CREATE TABLE IF NOT EXISTS group_task_inputs (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    task_id         TEXT NOT NULL REFERENCES group_tasks(id) ON DELETE CASCADE,
    message_id      TEXT NOT NULL,
    target_agent    TEXT NOT NULL,
    source_agent    TEXT NOT NULL DEFAULT 'user',
    model           TEXT NOT NULL,
    text            TEXT NOT NULL,
    status          TEXT NOT NULL DEFAULT 'queued',
    generation      INTEGER NOT NULL DEFAULT 0,
    attempt_id      TEXT,
    owner_instance  TEXT,
    owner_device_id TEXT REFERENCES devices(id) ON DELETE SET NULL,
    lease_until     BIGINT,
    retry_count     INTEGER NOT NULL DEFAULT 0,
    next_attempt_at BIGINT NOT NULL DEFAULT 0,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (task_id, message_id, target_agent)
);

CREATE INDEX IF NOT EXISTS idx_group_inputs_pending
    ON group_task_inputs (task_id, status, id);

CREATE TABLE IF NOT EXISTS group_task_steps (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    task_id         TEXT NOT NULL REFERENCES group_tasks(id) ON DELETE CASCADE,
    ordinal         INTEGER NOT NULL,
    agent           TEXT NOT NULL,
    source_agent    TEXT NOT NULL,
    source_message  TEXT,
    instruction     TEXT NOT NULL,
    status          TEXT NOT NULL DEFAULT 'running',
    sequence_start  BIGINT NOT NULL,
    sequence_end    BIGINT,
    before_snapshot JSONB NOT NULL DEFAULT '{}'::jsonb,
    after_snapshot  JSONB,
    messages        JSONB NOT NULL DEFAULT '[]'::jsonb,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    finished_at     TIMESTAMPTZ,
    UNIQUE (task_id, ordinal)
);

CREATE INDEX IF NOT EXISTS idx_group_task_steps_task
    ON group_task_steps (task_id, ordinal);

CREATE TABLE IF NOT EXISTS agent_attempts (
    id            TEXT PRIMARY KEY,
    input_id      BIGINT NOT NULL REFERENCES group_task_inputs(id) ON DELETE CASCADE,
    generation    INTEGER NOT NULL,
    phase         TEXT NOT NULL,
    native_run_id TEXT,
    error         TEXT,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    finished_at   TIMESTAMPTZ
);

-- ----------------------------------------------------------------------------
-- 6. 偏好设置与设备运行控制表
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS reasoning_preferences (
    conversation_id TEXT NOT NULL,
    agent           TEXT NOT NULL,
    model           TEXT NOT NULL,
    effort          TEXT NOT NULL,
    PRIMARY KEY (conversation_id, agent, model)
);

CREATE TABLE IF NOT EXISTS speed_preferences (
    conversation_id TEXT NOT NULL,
    model           TEXT NOT NULL,
    service_tier    TEXT NOT NULL,
    PRIMARY KEY (conversation_id, model)
);

-- 按设备隔离运行控制状态（替代原 SQLite 单行 id=1）
CREATE TABLE IF NOT EXISTS app_controls (
    device_id   TEXT PRIMARY KEY REFERENCES devices(id) ON DELETE CASCADE,
    paused      BOOLEAN NOT NULL DEFAULT FALSE,
    instance_id TEXT NOT NULL DEFAULT '',
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ----------------------------------------------------------------------------
-- 7. 跨端图片与附件存储表
-- ----------------------------------------------------------------------------

CREATE TABLE IF NOT EXISTS attachments (
    id                   TEXT PRIMARY KEY,
    project_id           TEXT REFERENCES projects(id) ON DELETE SET NULL,
    conversation_id      TEXT REFERENCES conversations(id) ON DELETE SET NULL,
    file_name            TEXT NOT NULL,
    mime_type            TEXT NOT NULL,
    byte_size            BIGINT NOT NULL,
    sha256               TEXT NOT NULL,
    data                 BYTEA NOT NULL,
    created_by_device_id TEXT REFERENCES devices(id) ON DELETE SET NULL,
    created_at           TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_attachments_sha256
    ON attachments (sha256);

CREATE INDEX IF NOT EXISTS idx_attachments_conversation
    ON attachments (conversation_id, created_at);

-- ----------------------------------------------------------------------------
-- 8. 触发器：自动同步 agent_sessions 运行时消息至主会话
-- ----------------------------------------------------------------------------

CREATE OR REPLACE FUNCTION fn_agent_message_insert()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO codex_messages (conversation_id, item_id, role, text, created_at, agent)
    SELECT s.conversation_id, NEW.item_id, NEW.role, NEW.text, NEW.created_at, s.agent_id
    FROM agent_sessions s
    WHERE s.runtime_conversation_id = NEW.conversation_id
    ON CONFLICT (conversation_id, item_id)
    DO UPDATE SET text = EXCLUDED.text, role = EXCLUDED.role;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_agent_message_insert ON codex_messages;
CREATE TRIGGER trg_agent_message_insert
AFTER INSERT ON codex_messages
FOR EACH ROW
EXECUTE FUNCTION fn_agent_message_insert();

CREATE OR REPLACE FUNCTION fn_agent_message_update()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE codex_messages m
    SET text = NEW.text,
        role = NEW.role
    FROM agent_sessions s
    WHERE s.runtime_conversation_id = NEW.conversation_id
      AND m.conversation_id = s.conversation_id
      AND m.item_id = NEW.item_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_agent_message_update ON codex_messages;
CREATE TRIGGER trg_agent_message_update
AFTER UPDATE OF text, role ON codex_messages
FOR EACH ROW
EXECUTE FUNCTION fn_agent_message_update();
```
